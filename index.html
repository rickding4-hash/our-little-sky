首次设置

Run this once in Supabase → SQL Editor. It creates the member table, locks writes to the administrator only, and gives both of you read access.

Show setup SQL
-- 1) who belongs here, and with which role
create table if not exists public.couple_members (
  user_id uuid primary key references auth.users(id) on delete cascade,
  role text not null default 'viewer' check (role in ('admin','viewer')),
  created_at timestamptz not null default now()
);
alter table public.couple_members enable row level security;
drop policy if exists "read own role" on public.couple_members;
create policy "read own role" on public.couple_members
  for select to authenticated using (auth.uid() = user_id);

-- 2) shared memories
create table if not exists public.memories (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  memory_date date,
  story text default '',
  photo_paths text[] not null default '{}',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);
alter table public.memories enable row level security;

-- 3) role helper: is the caller the administrator?
create or replace function public.is_couple_admin() returns boolean
language sql stable security definer set search_path = public as $$
  select exists (
    select 1 from public.couple_members m
    where m.user_id = auth.uid() and m.role = 'admin'
  );
$$;
revoke all on function public.is_couple_admin() from public;
grant execute on function public.is_couple_admin() to authenticated;

-- 4) memories: both read, only the admin writes
drop policy if exists "Members can read memories" on public.memories;
drop policy if exists "Members can add memories" on public.memories;
drop policy if exists "Members can edit memories" on public.memories;
drop policy if exists "Members can delete memories" on public.memories;
create policy "Members can read memories" on public.memories
  for select to authenticated using (true);
create policy "Admin can add memories" on public.memories
  for insert to authenticated with check (public.is_couple_admin());
create policy "Admin can edit memories" on public.memories
  for update to authenticated using (public.is_couple_admin()) with check (public.is_couple_admin());
create policy "Admin can delete memories" on public.memories
  for delete to authenticated using (public.is_couple_admin());

-- 5) private photo bucket: both read, only the admin writes
insert into storage.buckets (id, name, public) values ('couple-memories', 'couple-memories', false) on conflict (id) do nothing;
drop policy if exists "Members can read couple photos" on storage.objects;
drop policy if exists "Members can upload couple photos" on storage.objects;
drop policy if exists "Members can update couple photos" on storage.objects;
drop policy if exists "Members can delete couple photos" on storage.objects;
create policy "Members can read couple photos" on storage.objects
  for select to authenticated using (bucket_id = 'couple-memories');
create policy "Admin can upload couple photos" on storage.objects
  for insert to authenticated with check (bucket_id = 'couple-memories' and public.is_couple_admin());
create policy "Admin can update couple photos" on storage.objects
  for update to authenticated using (bucket_id = 'couple-memories' and public.is_couple_admin())
  with check (bucket_id = 'couple-memories' and public.is_couple_admin());
create policy "Admin can delete couple photos" on storage.objects
  for delete to authenticated using (bucket_id = 'couple-memories' and public.is_couple_admin());

-- 6) register the two accounts (replace the emails), then disable sign-ups
insert into public.couple_members (user_id, role)
select id, 'admin' from auth.users where email = 'you@example.com'
on conflict (user_id) do update set role = 'admin';
insert into public.couple_members (user_id, role)
select id, 'viewer' from auth.users where email = 'her@example.com'
on conflict (user_id) do update set role = 'viewer';
